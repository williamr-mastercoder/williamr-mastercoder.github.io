<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>My Private Music Player</title>
</head>
<body>
  <h1>My Music</h1>

  <button onclick="playRandom()">Shuffle</button>
  <button onclick="toggleRepeat()">Repeat: Off</button>
  <br><br>

  <audio id="player" controls></audio>
  <br><br>

  <ul id="playlist"></ul>

  <script>
    const songs = [
      "music/01 - Preacher Man.mp3",
      "music/LOVE ME THE MOST.mp3",
      "music/NITROUS_L0VE.mp3"
    ];

    const player = document.getElementById("player");
    const playlist = document.getElementById("playlist");
    let repeat = false;
    let currentIndex = 0;

    // build playlist on screen
    songs.forEach((song, index) => {
      const li = document.createElement("li");
      const btn = document.createElement("button");
      btn.textContent = "Play " + song.split("/")[1];
      btn.onclick = () => playSong(index);
      li.appendChild(btn);
      playlist.appendChild(li);
    });

    function playSong(index) {
      currentIndex = index;
      player.src = songs[index];
      player.play();
    }

    function playRandom() {
      const randomIndex = Math.floor(Math.random() * songs.length);
      playSong(randomIndex);
    }

    function toggleRepeat() {
      repeat = !repeat;
      document.querySelector("button[onclick='toggleRepeat()']")
        .textContent = "Repeat: " + (repeat ? "On" : "Off");
    }

    // when a song ends
    player.onended = () => {
      if (repeat) {
        playSong(currentIndex);
      } else {
        playRandom();
      }
    };

    // start with first song loaded
    playSong(0);
  </script>
</body>
</html>
